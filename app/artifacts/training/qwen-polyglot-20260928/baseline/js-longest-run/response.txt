const { strict as assert } = require('assert');

function solve(input) {
    if (input.length === 0) return 0;
    let maxLength = 1;
    let currentLength = 1;
    for (let i = 1; i < input.length; i++) {
        if (input[i] === input[i - 1]) {
            currentLength++;
        } else {
            maxLength = Math.max(maxLength, currentLength);
            currentLength = 1;
        }
    }
    maxLength = Math.max(maxLength, currentLength);
    return maxLength;
}

module.exports = solve;
