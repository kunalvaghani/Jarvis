def flatten_list(data):
    # Base case: if the data is a list, return it as is
    if isinstance(data, list):
        return data
    
    # Recursive case: flatten each element in the list
    result = []
    for item in data:
        if isinstance(item, list):
            result.extend(flatten_list(item))
        else:
            result.append(item)
    
    return result

    main()
