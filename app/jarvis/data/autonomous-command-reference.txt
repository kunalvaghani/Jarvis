8. In-File Code M anipulation (Part 1)
U sing echo and stream  editors (sed) to inject, m odify, and delete code logic directly from  the term inal
without graphical editors.
Python
# 1. Add function
echo -e "\ndef greet(name):\n    return f'Hello {name}'" >> main.py
# 2. Call function
echo "print(greet('Agent'))" >> main.py
# 3. Edit function logic
sed -i 's/Hello/Welcome/g' main.py
# 4. Delete specific line
sed -i '/print(greet/d' main.py
JavaScript (Node.js)
# 1. Add function
echo -e "\nfunction calc(r) {\n  return Math.PI * r;\n}" >> index.js
# 2. Call function
echo "console.log(calc(5));" >> index.js
# 3. Edit logic
sed -i 's/Math.PI \* r/Math.PI \* r \* r/g' index.js
# 4. Delete block (from function declaration to closing brace)
sed -i '/function calc/,/}/d' index.js
TypeScript
# 1. Add typed arrow function
echo -e "\nconst multiply = (a: number, b: number): number => a * b;" >> index.ts
# 2. Call function
echo "console.log(multiply(10, 5));" >> index.ts
# 3. Edit logic (multiply to divide)
sed -i 's/a \* b/a \/ b/g' index.ts
# 4. Delete execution line
sed -i '/console.log(multiply/d' index.ts
React (JSX)
# 1. Add component
echo -e "\nexport const ActionBtn = () => <button>Click</button>;" >> App.jsx
# 2. Inject component inside existing structure before closing div
sed -i 's/<\/div>/  <ActionBtn \/>\n<\/div>/' App.jsx
# 3. Edit component string
sed -i 's/Click/Submit/g' App.jsx
# 4. Delete component usage
sed -i '/<ActionBtn \/>/d' App.jsx
Rust
# 1. Add function outside main
echo -e "\nfn alert() {\n    println!(\"System ready.\");\n}" >> src/main.rs
# 2. Inject call immediately after fn main() {
sed -i '/fn main() {/a \    alert();' src/main.rs
# 3. Edit string
sed -i 's/"System ready."/"Agent online."/' src/main.rs
# 4. Delete call
sed -i '/alert();/d' src/main.rs

9. In-File Code M anipulation (Part 2)
C / C++
# 1. Insert function at line 1 (avoiding header prototype issues)
sed -i '1i void init() { printf("Init\\n"); }' src/main.cpp
# 2. Inject right after int main() {
sed -i '/int main/a \    init();' src/main.cpp
# 3. Edit string
sed -i 's/"Init\\n"/"Boot\\n"/' src/main.cpp
# 4. Delete definition entirely
sed -i '/void init()/,/}/d' src/main.cpp
Java
# 1. Add method before the final closing brace (^)
sed -i '/^}$/i \    public static void ping() { System.out.println("Pong"); }' App.java
# 2. Inject right after public static void main
sed -i '/public static void main/a \        ping();' App.java
# 3. Edit print statement
sed -i 's/"Pong"/"Connected"/' App.java
# 4. Delete execution line
sed -i '/ping();/d' App.java
C# (.NET Top-L evel Statem ents)
# 1. Add local function
echo -e "\nstring GetStatus(int c) => c == 200 ? \"OK\" : \"Err\";" >> Program.cs
# 2. Call function
echo "Console.WriteLine(GetStatus(200));" >> Program.cs
# 3. Edit ternary return
sed -i 's/"OK"/"Success"/' Program.cs
# 4. Delete usage
sed -i '/Console.WriteLine(GetStatus/d' Program.cs
Go
# 1. Add function
echo -e "\nfunc cleanup() {\n\tfmt.Println(\"Clean\")\n}" >> main.go
# 2. Match 'func main', skip to closing '}', insert before it
sed -i '/func main()/,/^}/ s/^}/\tcleanup()\n}/' main.go
# 3. Edit string
sed -i 's/"Clean"/"Wipe"/g' main.go
# 4. Delete call
sed -i '/cleanup()/d' main.go
